#include <tunables/global>

profile claudecode flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Capabilities needed for terminal and networking
  capability net_admin,
  capability net_raw,
  capability setgid,
  capability setuid,
  capability chown,
  capability fowner,
  capability dac_override,
  capability dac_read_search,
  capability sys_tty_config,

  # Network access for Claude API and hass-mcp
  network,

  # PTY/TTY access for terminal
  /dev/ptmx rw,
  /dev/pts/* rw,
  /dev/tty* rw,
  /dev/null rw,
  /dev/zero r,
  /dev/urandom r,

  # System files (read-only)
  /etc/** r,
  /usr/** r,
  /lib/** r,
  /proc/** r,
  /sys/** r,

  # Execute permissions for binaries
  /bin/** ixr,
  /usr/bin/** ixr,
  /usr/lib/** ixmr,
  /usr/local/** ixrw,
  /usr/libexec/** ixr,

  # Add-on data
  /data/** rwk,

  # Home Assistant config - main working directory
  /homeassistant/ r,
  /homeassistant/** rwk,

  # MemSearch lives on persistent storage under /homeassistant/.claudecode.
  # Its venv (python, onnxruntime .so, bundled milvus-lite binary) must be
  # executable/mmap-able, and onnxruntime mmaps the downloaded ONNX model.
  /homeassistant/.claudecode/memsearch-venv/** ixmr,
  /homeassistant/.claudecode/hf-cache/** rm,

  # Other mapped HA directories
  /ssl/ r,
  /ssl/** r,
  /share/ r,
  /share/** rwk,
  /media/ r,
  /media/** rwk,
  /backup/ r,
  /backup/** r,
  /addon_configs/ r,
  /addon_configs/** rwk,
  /config/ r,
  /config/** rwk,

  # Claude Code and user home
  /root/ r,
  /root/** rwk,
  /tmp/** rwk,

  # Node.js modules
  /usr/lib/node_modules/** ixmr,

  # Debian ships /usr/bin/npm as a symlink to /usr/share/nodejs/npm/bin/npm-cli.js.
  # AppArmor resolves symlinks before checking, and /usr/share only had `r` from
  # the /usr/** rule above — so execve() on npm returned EACCES and bash reported
  # it as "/usr/bin/env: bad interpreter: Permission denied". That silently broke
  # auto-update on every boot. The real path needs exec/mmap.
  /usr/share/nodejs/** ixmr,

  # Playwright MCP server (installed under /opt by the Dockerfile, launched via `node`)
  /opt/ r,
  /opt/** ixmr,

  # code-server (ui_mode: vscode) keeps user-data + extensions in /data/vscode
  # (covered rwk by /data/** above). Extensions may ship native .node addons and
  # helper binaries, so that tree additionally needs exec/mmap.
  /data/vscode/** ixmr,
}
