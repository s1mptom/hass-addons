#include <tunables/global>

profile claudecode flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Capabilities needed for terminal and networking
  capability net_admin,
  capability net_raw,
  capability setgid,
  capability setuid,
  capability chown,
  capability fowner,
  capability dac_override,
  capability dac_read_search,
  capability sys_tty_config,

  # Network access for Claude API and hass-mcp
  network,

  # PTY/TTY access for terminal
  /dev/ptmx rw,
  /dev/pts/* rw,
  /dev/tty* rw,
  /dev/null rw,
  /dev/zero r,
  /dev/urandom r,

  # System files (read-only)
  /etc/** r,
  /usr/** r,
  /lib/** r,
  /proc/** r,
  /sys/** r,

  # Execute permissions for binaries
  /bin/** ixr,
  /usr/bin/** ixr,
  /usr/lib/** ixmr,
  /usr/local/** ixrwl,
  /usr/libexec/** ixr,

  # Add-on data
  /data/** rwkl,

  # Home Assistant config - main working directory
  /homeassistant/ r,
  /homeassistant/** rwkl,

  # MemSearch lives on persistent storage under /homeassistant/.claudecode.
  # Its venv (python, onnxruntime .so, bundled milvus-lite binary) must be
  # executable/mmap-able, and onnxruntime mmaps the downloaded ONNX model.
  /homeassistant/.claudecode/memsearch-venv/** ixmr,
  /homeassistant/.claudecode/hf-cache/** rm,

  # Other mapped HA directories
  /ssl/ r,
  /ssl/** r,
  /share/ r,
  /share/** rwk,
  /media/ r,
  /media/** rwk,
  /backup/ r,
  /backup/** r,
  /addon_configs/ r,
  /addon_configs/** rwk,
  /config/ r,
  /config/** rwk,

  # Claude Code and user home.
  # NOTE the `l` (link) permission: AppArmor treats hardlink creation as its own
  # permission, separate from write. npm's cache (cacache) hardlinks downloads
  # from _cacache/tmp into _cacache/content-v2, so without `l` every install
  # fails with "EACCES / syscall link" — which npm unhelpfully reports as a
  # root-owned-cache problem. Same for pip/venv under the paths below.
  # `ix` matters as much as rwkl here: Claude Code plugins, skills and anything
  # dropped in /root/.local/bin live under /root, and plugin hooks execute helper
  # scripts *directly* (e.g. MemSearch's stop.sh runs "$SCRIPT_DIR/parse-transcript.sh").
  # Without exec those calls fail with "Permission denied" even though the file
  # has its exec bit — and because such hooks routinely swallow errors with
  # `2>/dev/null || true`, the feature silently does nothing. That is exactly how
  # MemSearch recorded empty session headers with no turn summaries for weeks.
  /root/ r,
  /root/** ixrwkl,
  /tmp/** rwkl,

  # Node.js modules
  /usr/lib/node_modules/** ixmr,

  # Kept as a safety net: since 1.4.6 Node/npm come from the official tarball in
  # /usr/local (covered above), but if Debian's nodejs/npm ever gets pulled in as
  # a dependency again, this is what it needs. Original problem was:
  # Debian ships /usr/bin/npm as a symlink to /usr/share/nodejs/npm/bin/npm-cli.js.
  # AppArmor resolves symlinks before checking, and /usr/share only had `r` from
  # the /usr/** rule above — so execve() on npm returned EACCES and bash reported
  # it as "/usr/bin/env: bad interpreter: Permission denied". That silently broke
  # auto-update on every boot. The real path needs exec/mmap.
  /usr/share/nodejs/** ixmr,

  # Playwright MCP server (installed under /opt by the Dockerfile, launched via `node`)
  /opt/ r,
  /opt/** ixmr,

  # code-server (ui_mode: vscode) keeps user-data + extensions in /data/vscode
  # (covered rwk by /data/** above). Extensions may ship native .node addons and
  # helper binaries, so that tree additionally needs exec/mmap.
  /data/vscode/** ixmr,
}
